What is GitOps?

GitOps is a way of managing infrastructure and applications using Git as the single source of truth.

Simple idea: Make changes in Git → Automatically applied to system

Why GitOps?

Automation

Version control

Easy rollback

Transparency

Faster deployments

Real Example

Instead of manually deploying apps:

Update code in Git → System auto deploys changes



Core Principles

Git is the source of truth

Automated deployment

Continuous monitoring

Example Flow

Developer → Push code → Git → Tool detects change → Deploy

GitLab Configuration Sample

# --------------------------------------------------

$ GitLab CI/CD Pipeline with Explanation
$ --------------------------------------------------
$ Variables (Used across all stages)
variables:

IMAGE_NAME: demo/gitlab # Docker image name

IMAGE_TAG: v2.0 # Docker image version/tag

$ Define pipeline stages
stages:

- install

- test

- build

- image_build

- deploy

$ --------------------------------------------------
$ Stage 1: Install Dependencies
$ --------------------------------------------------

install:

image: node:14-alpine # Use Node.js lightweight image

stage: install

script:

- npm install # Install project dependencies

$ Cache to speed up future runs

cache:

paths:

- node_modules/

$ Share files with next stages

artifacts:

paths:

- node_modules/

$ --------------------------------------------------
$ Stage 2: Testing
$ --------------------------------------------------

test:

image: node:14-alpine

stage: test

before_script:

- npm install react-scripts --save # Install testing tools

script:

- npm test # Run test cases

$ --------------------------------------------------
$ Stage 3: Build Application
$ --------------------------------------------------

build:

image: node:14-alpine

stage: build

script:

- npm run build # Build production-ready app

artifacts:

paths:

- node_modules/

$ --------------------------------------------------
$ Stage 4: Docker Image Build & Push
$ --------------------------------------------------

image_build:

image: docker:27.3

stage: image_build

services:

- docker:27.3-dind # Docker-in-Docker service

before_script:

- docker login -u $DOCKER_USER -p $DOCKER_PASS # Login to Docker Hub

script:

- docker build -t $IMAGE_NAME:$IMAGE_TAG . # Build Docker image

- docker push $IMAGE_NAME:$IMAGE_TAG # Push image to registry

$ --------------------------------------------------
$ Stage 5: Deployment
$ --------------------------------------------------

deploy:

stage: deploy

image: docker:27.3

before_script:

- chmod 400 $SSH_KEY # Secure SSH key permissions

script:

- ssh -i $SSH_KEY -o StrictHostKeyChecking=no $AWS_EC2_USER@$AWS_EC2_HOST "

$ Stop and remove existing containers
$ docker ps -a -q | xargs -r docker stop | xargs -r docker rm;
$ Run new container with latest image
$ docker run -d --name demo -p 80:3000 $IMAGE_NAME:$IMAGE_TAG

"
